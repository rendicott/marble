1. Behavior (locked)

# Local
marble-harness --base-url http://127.0.0.1:8000/v1   # no Authorization

# Cloud
export OPENAI_API_KEY=sk-...
marble-harness --base-url https://api.openai.com/v1 \
  --api-key-env=OPENAI_API_KEY                       # Authorization: Bearer $KEY

# Fallback env names
--api-key-env=OPENAI_API_KEY,OPENROUTER_API_KEY

# systemd
EnvironmentFile=-%h/.config/marble/model.env   # mode 0600
ExecStart=... --api-key-env=OPENAI_API_KEY ...
